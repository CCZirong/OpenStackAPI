import json
import yaml
import requests


class K8SServiceManager:
    def __init__(self, api_server, api_token, namespace="default"):
        self.api_server = api_server
        self.api_token = api_token
        self.namespace = namespace
        self.headers = {
            "Authorization": f"Bearer {self.api_token}",
            "Content-Type": "application/json"
        }

    def send_request(self, method, url, data=None):
        try:
            response = requests.request(method, url, headers=self.headers, json=data, verify=False)
            response.raise_for_status()
            return response.json()
        except requests.exceptions.RequestException as e:
            print(f"Error: {e}")
            return None

    def delete_service_if_exists(self, service_name):
        url = f"{self.api_server}/api/v1/namespaces/{self.namespace}/services/{service_name}"
        response = self.send_request("GET", url)
        if response:
            print(f"Service '{service_name}' exists. Deleting...")
            self.send_request("DELETE", url)
            print(f"Service '{service_name}' deleted.")

    def create_service(self, service_yaml):
        with open(service_yaml, "r") as f:
            service_data = yaml.safe_load(f)
        url = f"{self.api_server}/api/v1/namespaces/{self.namespace}/services"
        response = self.send_request("POST", url, service_data)
        if response:
            print(f"Service '{service_data['metadata']['name']}' created successfully.")
        return response

    def get_service_info(self, service_name):
        url = f"{self.api_server}/api/v1/namespaces/{self.namespace}/services/{service_name}"
        return self.send_request("GET", url)

    def update_service(self, service_name, service_update_yaml):
        with open(service_update_yaml, "r") as f:
            update_data = yaml.safe_load(f)
        url = f"{self.api_server}/api/v1/namespaces/{self.namespace}/services/{service_name}"
        response = self.send_request("PUT", url, update_data)
        if response:
            print(f"Service '{service_name}' updated successfully.")
        return response

    def manage_service(self, service_name, service_yaml, service_update_yaml, output_file):
        self.delete_service_if_exists(service_name)

        self.create_service(service_yaml)

        service_info = self.get_service_info(service_name)
        if service_info:
            with open(output_file, "w") as f:
                json.dump(service_info, f, indent=4)
            print(f"Service info saved to {output_file}.")

        self.update_service(service_name, service_update_yaml)

        updated_service_info = self.get_service_info(service_name)
        if updated_service_info:
            print("Updated Service Info:")
            print(json.dumps(updated_service_info, indent=4))
            with open(output_file, "a") as f:
                f.write("\n")
                json.dump(updated_service_info, f, indent=4)

```shell
from kubernetes import client, config, watch
import random
import json


# 1.加载config
config.load_kube_config()


# 2.初始化客户端
v1 = client.CoreV1Api()

# 3.监听资源
w = watch.Watch()

# 获取所有node节点
#print(type(v1.list_node().items))
#print(dir(random))

#print(dir(client))
# print(dir(v1))
# 4.调度实现
def schduler(pod_name, node_name, namespace):
  # 4.1 构建调度的节点的信息
  target = client.V1ObjectReference(kind="Node", api_version="v1", name=node_name)
  meta = client.V1ObjectMeta()
  meta.name = pod_name
  # 4.2 创建调度绑定(绑定到对应节点上)
  body = client.V1Binding(target=target)
  # 4.3 调度将Pending Pod调度到对应namespace上
  body.metadata = meta 
  try:
    v1.create_namespaced_binding(body=body, namespace=namespace)
  except Exception as e:
    return
  



for event in w.stream(v1.list_namespaced_pod, namespace='default'):
  pod = event['object']
  if pod.status.phase == "Pending" or pod.status.phase == "Running":
    # 随机选取一个节点 并抽取出名称
    node_name = random.choice(v1.list_node().items).metadata.name
    # 实现调度
    schduler(pod.metadata.name, node_name, pod.metadata.namespace)
    # 打印输出
    resp = {"type": event['type'], 
            "pod_name": pod.metadata.name, 
            "node_name": node_name, 
            "schduler_name": pod.spec.scheduler_name
           }
    print(json.dumps(resp)) 
    exit()
```

#### Python自动化--使用SDK方式进行RBAC管理

使用Python SDK方式进行RBAC管理，要求如下：

（1）使用Kubernetes python SDK的“kubernetes”Python库，在/root目录下，创建rbac.py文件，要求编写python代码，代码实现以下任务：

（2）创建ServiceAccount资源，名为sa-resources；

（3）创建ClusterRole资源，名为clusterrole-resources；

（4）创建ClusterRoleBinding资源，名为binding-resources，并将集群角色clusterrole-resources与sa-resources进行绑定；请准备好测试环境。

```shell
from kubernetes import client, config
import json

# 加载 Kubernetes 配置
config.load_kube_config()

# 创建 Kubernetes 客户端对象
v1 = client.CoreV1Api()
rbac = client.RbacAuthorizationV1Api()

# 创建 ServiceAccount 资源
sa = client.V1ServiceAccount(
    metadata=client.V1ObjectMeta(name="sa-resources"),
)
v1.create_namespaced_service_account(namespace="default", body=sa)
print(v1.read_namespaced_service_account(namespace="default", name="sa-resources"))

# 创建 ClusterRole 资源
cr = client.V1ClusterRole(
    metadata=client.V1ObjectMeta(name="clusterrole-resources"),
    rules=[
        client.V1PolicyRule(
            api_groups=["extensions", "apps"],
            resources=["deployments"],
            verbs=["get", "watch", "list", "create", "update", "patch", "delete"],
        ),
    ],
)
rbac.create_cluster_role(body=cr)
print(rbac.read_cluster_role(name="clusterrole-resources"))

# 创建 ClusterRoleBinding 资源
crb = client.V1ClusterRoleBinding(
    metadata=client.V1ObjectMeta(name="binding-resources"),
    role_ref=client.V1RoleRef(
        api_group="rbac.authorization.k8s.io",
        kind="ClusterRole",
        name="clusterrole-resources",
    ),
    subjects=[client.V1Subject(
        kind="ServiceAccount",
        name="sa-resources",
        namespace="default",
    )],
)
rbac.create_cluster_role_binding(body=crb)
print(rbac.read_cluster_role_binding(name="binding-resources"))
```

def get_random_node():

    nodes_url = f"{KUBERNETES_API_SERVER}/api/v1/nodes"
    response = requests.get(nodes_url, headers=headers, verify=False)

                available_nodes.append(i["metadata"]["name"])

    # 随机选择一个节点

        },
        "target": {
            "apiVersion": "v1",
            "kind": "Node",
            "name": node_name,
        }
    }
